<script>
const arena = document.getElementById("arena");
const scoreBox = document.getElementById("score");
const comboBox = document.getElementById("combo");
const timeBox = document.getElementById("time");
const overlay = document.getElementById("overlay");
const final = document.getElementById("final");

let score, combo, timeLeft, speed, timerTick, targetTimer;
const AudioCtx = window.AudioContext || window.webkitAudioContext;
const audio = new AudioCtx();

function beep(freq,duration){
    const o = audio.createOscillator();
    const g = audio.createGain();
    o.frequency.value=freq;
    o.connect(g);
    g.connect(audio.destination);
    g.gain.value=.08;
    o.start();
    o.stop(audio.currentTime+duration);
}

function spawn(){
    const t = document.createElement("div");
    t.className="target";
    const size=70;
    t.style.left=Math.random()*(innerWidth-size)+"px";
    t.style.top=Math.random()*(innerHeight-size)+"px";

    t.onclick=e=>{
        e.stopPropagation();
        beep(600+combo*40,.08);
        score+=10*combo;
        combo++;
        speed*=0.94;
        updateHUD();
        clearTimeout(targetTimer);
        t.remove();
        spawn();
    };

    arena.appendChild(t);

    targetTimer=setTimeout(()=>{
        combo=1;
        t.remove();
        spawn();
    },speed);
}

function updateHUD(){
    scoreBox.textContent="Score: "+score;
    comboBox.textContent="Combo: x"+combo;
}

function start(){
    arena.innerHTML="";
    overlay.style.display="none";
    score=0;
    combo=1;
    timeLeft=60;
    speed=1200;
    updateHUD();
    timeBox.textContent="Time: 60.0";

    timerTick=setInterval(()=>{
        timeLeft-=0.1;
        timeBox.textContent="Time: "+timeLeft.toFixed(1);
        if(timeLeft<=0) end();
    },100);

    spawn();
}

function end(){
    clearInterval(timerTick);
    clearTimeout(targetTimer);
    arena.innerHTML="";
    final.textContent="Final Score: "+score;
    overlay.style.display="flex";
    beep(180,.4);
}

document.body.addEventListener("click",()=>{
    combo=1;
    updateHUD();
});

start();
</script>
